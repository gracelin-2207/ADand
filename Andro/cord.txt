package com.example.ex4;

import android.os.Bundle;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

@Override
protected void onCreate(Bundle b) {
super.onCreate(b);

WebView web = new WebView(this);
setContentView(web);

// Enable JavaScript
web.getSettings().setJavaScriptEnabled(true);

// Open inside app (not browser)
web.setWebViewClient(new WebViewClient());

// Load HTML file
web.loadUrl("file:///android_asset/index.html");
}
}

<p id="loc"></p>
<p id="result"></p>

<script>

// Step 1: Get location
function getLocation() {
    navigator.geolocation.getCurrentPosition(showPosition);
}

// Step 2: Show location
function showPosition(position) {

    let lat = position.coords.latitude;
    let lon = position.coords.longitude;

    document.getElementById("loc").innerHTML =
        "Lat: " + lat + "<br>Lon: " + lon;

    findPlaces(lat, lon);
}

// Step 3: Find nearby places
function findPlaces(lat, lon) {

    let query = `
    [out:json];
    (
      node["amenity"="hospital"](around:1000, ${lat}, ${lon});
      node["amenity"="atm"](around:1000, ${lat}, ${lon});
      node["amenity"="restaurant"](around:1000, ${lat}, ${lon});
    );
    out;
    `;

    let url = "https://overpass-api.de/api/interpreter?data=" + encodeURIComponent(query);

    fetch(url)
    .then(res => res.json())
    .then(data => {

        let output = "";

        data.elements.forEach(p => {
            output += (p.tags.name || "No Name") + " (" + p.tags.amenity + ")<br>";
        });

        document.getElementById("result").innerHTML = output;
    });
}

</script>

</body>
</html>
