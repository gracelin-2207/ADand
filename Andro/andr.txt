package com.example.calculator;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

   TextView t;

   double firstNumber = 0;
   String operator = "";

   @Override
   protected void onCreate(Bundle b) {
       super.onCreate(b);
       setContentView(R.layout.activity_main);

       t = findViewById(R.id.t);
   }

   // This method is called when ANY button is clicked
   public void press(View v) {

       Button btn = (Button) v;
       String value = btn.getText().toString();
       String current = t.getText().toString();

       // 1. If number is pressed
       if (value.matches("[0-9]")) {
           if (current.equals("0")) {
               t.setText(value);
           } else {
               t.setText(current + value);
           }
       }

       // 2. If operator is pressed
       else if (value.equals("+") || value.equals("-") ||
                value.equals("*") || value.equals("/")) {

           firstNumber = Double.parseDouble(current);
           operator = value;
           t.setText("0");
       }

       // 3. If equal is pressed
       else if (value.equals("=")) {

           double secondNumber = Double.parseDouble(current);
           double result = 0;

           if (operator.equals("+")) result = firstNumber + secondNumber;
           else if (operator.equals("-")) result = firstNumber - secondNumber;
           else if (operator.equals("*")) result = firstNumber * secondNumber;
           else if (operator.equals("/")) {
               if (secondNumber != 0)
                   result = firstNumber / secondNumber;
               else
                   result = 0;
           }

           t.setText("" + result);
       }

       // 4. If clear is pressed
       else if (value.equals("C")) {
           t.setText("0");
           firstNumber = 0;
           operator = "";
       }
   }
}

//DBHelper.java 

package com.example.ex2;

import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

public class DBHelper extends SQLiteOpenHelper {

   public DBHelper(Context context) {
       super(context, "StudentDB", null, 1);
   }

   @Override
   public void onCreate(SQLiteDatabase db) {
       // Create table
       db.execSQL("CREATE TABLE students(id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT)");
   }

   @Override
   public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
       db.execSQL("DROP TABLE IF EXISTS students");
       onCreate(db);
   }

   // Insert data
   public void insert(String name) {
       SQLiteDatabase db = getWritableDatabase();
       db.execSQL("INSERT INTO students(name) VALUES('" + name + "')");
   }

   // View data
   public Cursor view() {
       SQLiteDatabase db = getReadableDatabase();
       return db.rawQuery("SELECT * FROM students", null);
   }
}

package com.example.ex2;

import android.database.Cursor;
import android.os.Bundle;
import android.view.View;
import android.widget.EditText;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

   EditText name;
   DBHelper db;

   @Override
   protected void onCreate(Bundle savedInstanceState) {
       super.onCreate(savedInstanceState);
       setContentView(R.layout.activity_main);

       name = findViewById(R.id.name);
       db = new DBHelper(this);
   }

   // Insert button click
   public void insertData(View v) {

       String n = name.getText().toString();

       if (n.isEmpty()) {
           Toast.makeText(this, "Enter name first", Toast.LENGTH_SHORT).show();
           return;
       }

       db.insert(n);
       Toast.makeText(this, "Inserted", Toast.LENGTH_SHORT).show();

       name.setText(""); // clear box
   }

   // View button click
   public void viewData(View v) {

       Cursor c = db.view();

       if (c.getCount() == 0) {
           Toast.makeText(this, "No Data Found", Toast.LENGTH_SHORT).show();
           return;
       }

       String data = "";

       while (c.moveToNext()) {
           data += c.getInt(0) + " - " + c.getString(1) + "\n";
       }

       Toast.makeText(this, data, Toast.LENGTH_LONG).show();
   }
}

package com.example.library;

import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.os.Bundle;
import android.view.View;
import android.widget.*;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

   EditText book, student;
   TextView result;
   DBHelper db;

   @Override
   protected void onCreate(Bundle b) {
       super.onCreate(b);
       setContentView(R.layout.activity_main);

       book = findViewById(R.id.book);
       student = findViewById(R.id.student);
       result = findViewById(R.id.result);

       db = new DBHelper(this);
   }

   // 1. ADD BOOK
   public void addBook(View v) {

       String b = book.getText().toString();

       if (b.isEmpty()) {
           Toast.makeText(this, "Enter book name", Toast.LENGTH_SHORT).show();
           return;
       }

       SQLiteDatabase d = db.getWritableDatabase();
       d.execSQL("INSERT INTO books VALUES('" + b + "','Available','-')");

       Toast.makeText(this, "Book Added", Toast.LENGTH_SHORT).show();
   }

   // 2. LEND BOOK
   public void lendBook(View v) {
       update("Lent");
   }

   // 3. RESERVE BOOK
   public void reserveBook(View v) {
       update("Reserved");
   }

   // COMMON METHOD
   void update(String status) {

       String b = book.getText().toString();
       String s = student.getText().toString();

       SQLiteDatabase d = db.getWritableDatabase();

       d.execSQL("UPDATE books SET status='" + status +
               "', student='" + s + "' WHERE name='" + b + "'");

       Toast.makeText(this, status, Toast.LENGTH_SHORT).show();
   }

   // 4. VIEW BOOKS
   public void viewBooks(View v) {

       SQLiteDatabase d = db.getReadableDatabase();
       Cursor c = d.rawQuery("SELECT * FROM books", null);

       String data = "";

       while (c.moveToNext()) {
           data += "Book: " + c.getString(0) +
                   "\nStatus: " + c.getString(1) +
                   "\nStudent: " + c.getString(2) + "\n\n";
       }

       result.setText(data);
   }
}
